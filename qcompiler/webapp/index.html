Quick API Reference

      
from qcompiler import QuantumCircuit, compile

qc = QuantumCircuit(n_qubits, n_clbits, name="my_circuit")

# Single-qubit gates
qc.h(0)           # Hadamard
qc.x(0)           # Pauli-X
qc.rz(pi/4, 0)    # Rotation around Z
qc.t(0)           # T gate

# Two-qubit gates
qc.cx(0, 1)       # CNOT
qc.cz(0, 1)       # CZ
qc.swap(0, 1)     # SWAP

# Three-qubit gates
qc.ccx(0, 1, 2)   # Toffoli

# Measurement
qc.measure(0, 0)  # qubit 0 → clbit 0
qc.measure_all()  # measure all

# Compile
result = compile(qc, optimization_level=1)
print(result.qasm)           # OpenQASM 3 string
print(result.stats)          # gate counts, depth, ...
